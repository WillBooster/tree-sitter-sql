CREATE TABLE IF NOT EXISTS hr.modern_items (
  item_id NUMBER(12) GENERATED BY DEFAULT ON NULL AS IDENTITY (START WITH 1 INCREMENT BY 1 NOCACHE),
  sku VARCHAR2(64 CHAR) NOT NULL,
  label NVARCHAR2(100) DEFAULT ON NULL N'unnamed',
  weight BINARY_FLOAT,
  ratio BINARY_DOUBLE,
  created_at TIMESTAMP(6) WITH LOCAL TIME ZONE DEFAULT SYSTIMESTAMP,
  warranty INTERVAL YEAR(2) TO MONTH,
  sla INTERVAL DAY(2) TO SECOND(3),
  payload JSON,
  is_active BOOLEAN DEFAULT TRUE,
  embedding VECTOR(1536, FLOAT32),
  doc BLOB,
  gross NUMBER(10, 2),
  tax NUMBER(10, 2) GENERATED ALWAYS AS (gross * 0.1) VIRTUAL INVISIBLE,
  CONSTRAINT modern_items_pk PRIMARY KEY (item_id) USING INDEX TABLESPACE users ENABLE VALIDATE,
  CONSTRAINT modern_items_uk UNIQUE (sku) DEFERRABLE INITIALLY DEFERRED RELY,
  CONSTRAINT modern_items_ck CHECK (gross >= 0) ENABLE NOVALIDATE
)
SEGMENT CREATION IMMEDIATE
PCTFREE 10 INITRANS 2
TABLESPACE users
STORAGE (INITIAL 64K NEXT 1M MAXEXTENTS UNLIMITED)
LOB (doc) STORE AS SECUREFILE (ENABLE STORAGE IN ROW CHUNK 8K NOCACHE NOLOGGING)
ROW STORE COMPRESS ADVANCED
PARALLEL 4;

CREATE OR REPLACE PACKAGE pkg_orders AUTHID CURRENT_USER AS
  e_bad_qty EXCEPTION;
  FUNCTION calc_tax(p_amount NUMBER) RETURN NUMBER DETERMINISTIC;
  PROCEDURE process_batch(p_limit IN NUMBER DEFAULT 100);
END pkg_orders;
/

CREATE OR REPLACE PACKAGE BODY pkg_orders AS
  FUNCTION calc_tax(p_amount NUMBER) RETURN NUMBER DETERMINISTIC IS
  BEGIN
    RETURN p_amount * 0.1;
  END calc_tax;

  PROCEDURE process_batch(p_limit IN NUMBER DEFAULT 100) IS
    TYPE id_tab IS TABLE OF NUMBER INDEX BY PLS_INTEGER;
    v_ids id_tab;
    CURSOR c_open IS SELECT order_id FROM orders WHERE ROWNUM <= p_limit;
  BEGIN
    OPEN c_open;
    FETCH c_open BULK COLLECT INTO v_ids LIMIT p_limit;
    CLOSE c_open;

    IF v_ids.COUNT = 0 THEN
      RETURN;
    END IF;

    FORALL i IN 1..v_ids.COUNT SAVE EXCEPTIONS
      UPDATE orders SET status = 'DONE' WHERE order_id = v_ids(i);

    FOR idx IN 1..v_ids.COUNT LOOP
      EXIT WHEN v_ids(idx) < 0;
      DBMS_OUTPUT.PUT_LINE(p_limit);
    END LOOP;
  EXCEPTION
    WHEN OTHERS THEN
      RAISE;
  END process_batch;
END pkg_orders;
/

CREATE OR REPLACE TRIGGER trg_orders_biu
  BEFORE INSERT OR UPDATE OF status ON orders
  FOR EACH ROW
DECLARE
  v_now TIMESTAMP := SYSTIMESTAMP;
BEGIN
  IF :NEW.status IS NULL THEN
    :NEW.status := 'NEW';
  END IF;
  EXECUTE IMMEDIATE 'BEGIN NULL; END;';
END trg_orders_biu;
/

  FOR INSERT OR UPDATE ON employees
  COMPOUND TRIGGER
  v_count PLS_INTEGER := 0;

  BEGIN
    v_count := 0;
  END BEFORE STATEMENT;

  AFTER EACH ROW IS
