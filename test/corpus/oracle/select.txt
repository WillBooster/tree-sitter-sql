  FUNCTION bump(n NUMBER) RETURN NUMBER IS
  BEGIN
    RETURN n + 1;
  END;
  org_chart (emp_id, mgr_id, lvl) AS (
    SELECT emp_id, mgr_id, 1 FROM emp WHERE mgr_id IS NULL
    UNION ALL
    SELECT e.emp_id, e.mgr_id, o.lvl + 1 FROM emp e JOIN org_chart o ON e.mgr_id = o.emp_id
  )
  SEARCH DEPTH FIRST BY emp_id ASC NULLS LAST SET seq_col
  CYCLE emp_id SET is_cycle TO 'Y' DEFAULT 'N'
