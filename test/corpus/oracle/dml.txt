MERGE INTO bonuses b
USING (SELECT emp_id, sal, dept_id FROM employees) e
ON (b.emp_id = e.emp_id)
WHEN MATCHED THEN UPDATE
  SET b.bonus = b.bonus + e.sal * 0.1
  WHERE e.dept_id = 20
  DELETE WHERE e.sal > 50000
WHEN NOT MATCHED THEN INSERT (b.emp_id, b.bonus)
  VALUES (e.emp_id, e.sal * 0.05)
  WHERE e.sal <= 50000
LOG ERRORS INTO err$_bonuses REJECT LIMIT 10;
