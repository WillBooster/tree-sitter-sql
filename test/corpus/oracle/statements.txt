DECLARE
  v_total NUMBER := 0;
  v_step PLS_INTEGER := 1;
BEGIN
  WHILE v_step <= 5 LOOP
    CASE v_step
      WHEN 1 THEN
        v_total := v_total + 10;
      ELSE
        v_total := v_total + v_step;
    END CASE;
    v_step := v_step + 1;
  END LOOP;
  COMMIT;
EXCEPTION
  WHEN NO_DATA_FOUND OR TOO_MANY_ROWS THEN
    ROLLBACK;
END;
/

PL/SQL labels, GOTO, CONTINUE WHEN, dynamic ref cursor, and pipelined function
:language(oracle)
================================================================================

CREATE OR REPLACE FUNCTION gen_rows(p_max NUMBER) RETURN num_tab PIPELINED IS
  v_cur SYS_REFCURSOR;
  v_val NUMBER;
BEGIN
  OPEN v_cur FOR 'SELECT :1 FROM dual' USING p_max;
  FETCH v_cur INTO v_val;
  CLOSE v_cur;

  <<row_loop>>
  FOR i IN 1..p_max LOOP
    CONTINUE row_loop WHEN i = 2;
    PIPE ROW (i);
    IF i >= v_val THEN
      GOTO done_label;
    END IF;
  END LOOP row_loop;

  <<done_label>>
  RETURN;
END gen_rows;
/
