CREATE TEMPORARY TABLE IF NOT EXISTS orders (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  code CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NOT NULL UNIQUE KEY,
  status ENUM('new', 'paid', 'shipped') DEFAULT 'new' NOT NULL,
  flags SET('gift', 'express') DEFAULT '',
  amount DECIMAL(12, 2) UNSIGNED NOT NULL DEFAULT 0.00,
  tax DECIMAL(12, 2) GENERATED ALWAYS AS (amount * 0.1) STORED,
  total DECIMAL(12, 2) AS (amount + tax) VIRTUAL INVISIBLE,
  payload JSON DEFAULT (JSON_OBJECT()),
  coords POINT SRID 4326 NOT NULL,
  created_at TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  updated_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  customer_id BIGINT UNSIGNED NOT NULL,
  INDEX idx_customer (customer_id ASC) USING BTREE COMMENT 'customer lookup' VISIBLE,
  INDEX idx_func ((LOWER(code)) DESC),
  FULLTEXT KEY ft_code (code) WITH PARSER ngram,
  SPATIAL KEY sp_coords (coords),
  CONSTRAINT fk_orders_customer FOREIGN KEY (customer_id) REFERENCES customers (id) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT chk_amount CHECK (amount >= 0) ENFORCED,
  CHECK (total >= amount) NOT ENFORCED
) ENGINE = InnoDB, AUTO_INCREMENT = 1000, DEFAULT CHARSET = utf8mb4 COLLATE = utf8mb4_0900_ai_ci ROW_FORMAT = DYNAMIC COMMENT = 'orders table' COMPRESSION = 'zlib' ENCRYPTION = 'N';

CREATE DEFINER = CURRENT_USER TRIGGER trg_orders_bi
BEFORE INSERT ON orders FOR EACH ROW FOLLOWS other_trg
BEGIN
  IF NEW.amount < 0 THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Invalid amount';
  END IF;
END;

DROP TRIGGER IF EXISTS trg_orders_bi;

CREATE PROCEDURE proc_sync(IN p_limit INT, OUT p_done INT, INOUT p_tag VARCHAR(20))
LANGUAGE SQL NOT DETERMINISTIC MODIFIES SQL DATA SQL SECURITY DEFINER COMMENT 'sync'
proc_lbl: BEGIN
  DECLARE v_id INT DEFAULT p_limit + 1;
  DECLARE v_finished INT DEFAULT @init_done;
  DECLARE dup_key CONDITION FOR SQLSTATE '23000';
  DECLARE cur CURSOR FOR SELECT id FROM orders LIMIT p_limit;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_finished = 1;
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    GET STACKED DIAGNOSTICS CONDITION 1 @msg = MESSAGE_TEXT;
    RESIGNAL;
  END;

  OPEN cur;
  read_loop: LOOP
    FETCH NEXT FROM cur INTO v_id;
    IF v_finished = 1 THEN
      LEAVE read_loop;
    ELSEIF v_id = 0 THEN
      ITERATE read_loop;
    END IF;
  END LOOP read_loop;
  CLOSE cur;

  WHILE p_limit > 0 DO
    SET p_limit = p_limit - 1;
  END WHILE;

  REPEAT
    SET p_done = 1;
  UNTIL p_done = 1 END REPEAT;

  CASE p_done
    WHEN 1 THEN SET p_tag = 'ok';
    ELSE SET p_tag = 'none';
  END CASE;
END proc_lbl;

CREATE FUNCTION fn_double(n INT) RETURNS INT DETERMINISTIC CONTAINS SQL
BEGIN
  RETURN n * 2;
END;

ALTER PROCEDURE proc_sync COMMENT 'updated';
DROP PROCEDURE IF EXISTS proc_sync;
DROP FUNCTION IF EXISTS fn_double;
